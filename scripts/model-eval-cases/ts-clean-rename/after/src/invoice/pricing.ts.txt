export interface LineItem {
  sku: string;
  quantity: number;
  unitPriceCents: number;
}

export function calculateTotalCents(items: LineItem[]): number {
  return items.reduce((sum, item) => sum + item.quantity * item.unitPriceCents, 0);
}

export function largestItem(items: LineItem[]): LineItem | null {
  let best: LineItem | null = null;
  for (const item of items) {
    if (
      best === null ||
      item.quantity * item.unitPriceCents > best.quantity * best.unitPriceCents
    ) {
      best = item;
    }
  }
  return best;
}

export function formatQuote(items: LineItem[]): string {
  const totalCents: number = calculateTotalCents(items);
  const discountRate = totalCents > 100_000 ? 0.05 : 0;
  const due = Math.round(totalCents * (1 - discountRate));
  const suffix = discountRate > 0 ? " (5% bulk discount applied)" : "";
  return `Total: $${(due / 100).toFixed(2)}${suffix}`;
}

export function qualifiesForFreeShipping(items: LineItem[]): boolean {
  return calculateTotalCents(items) >= 50_000;
}

export function discountBucket(items: LineItem[]): string {
  const totalCents: number = calculateTotalCents(items);
  if (totalCents >= 500_000) return "platinum";
  if (totalCents >= 100_000) return "gold";
  if (totalCents >= 50_000) return "silver";
  return "standard";
}

export function quoteSummary(items: LineItem[]): string {
  const totalCents: number = calculateTotalCents(items);
  const perAverage = items.length > 0 ? totalCents / items.length : 0;
  return `${totalCents} cents across ${items.length} items (${perAverage.toFixed(0)} avg)`;
}
